   129→   129→function QuizTab() {
   130→   130→  const [scope, setScope] = useState('mixed');
   131→   131→  const [difficulty, setDifficulty] = useState('mixed');
   132→   132→  const [questions, setQuestions] = useState<QuizQuestion[] | null>(null);
   133→   133→  const [loading, setLoading] = useState(false);
   134→   134→  const [idx, setIdx] = useState(0);
   135→   135→  const [chosen, setChosen] = useState<number | null>(null);
   136→   136→  const [answers, setAnswers] = useState<{ q: string; chosen: number; correct: number; ok: boolean }[]>([]);
   137→   137→  const [finished, setFinished] = useState(false);
   138→   138→
   139→   139→  const generate = async () => {
   140→   140→    setLoading(true);
   141→   141→    setQuestions(null);
   142→   142→    setFinished(false);
   143→   143→    setIdx(0);
   144→   144→    setChosen(null);
   145→   145→    setAnswers([]);
   146→   146→    try {
   147→   147→      const isCategory = ['mixed', ...Object.keys(CATEGORY_META)].includes(scope);
   148→   148→      const res = await apiFetch('/api/quiz', {
   149→   149→        method: 'POST',
   150→   150→        body: JSON.stringify({
   151→   151→          skillSlug: isCategory ? undefined : scope,
   152→   152→          category: isCategory ? scope : undefined,
   153→   153→          difficulty,
   154→   154→          count: 5,
   155→   155→        }),
   156→   156→      });
   157→   157→      const data = await readJson<{ error?: string; questions?: QuizQuestion[] }>(res);
   158→   158→      if (!res.ok || data.error) throw new Error(data.error);
   159→   159→      setQuestions(data.questions || []);
   160→   160→    } catch {
   161→   161→      setQuestions([]);
   162→   162→    } finally {
   163→   163→      setLoading(false);
   164→   164→    }
   165→   165→  };
   166→   166→
   167→   167→  const submit = () => {
   168→   168→    if (chosen === null || !questions) return;
   169→   169→    const q = questions[idx];
   170→   170→    const ok = chosen === q.answerIndex;
   171→   171→    setAnswers(a => [...a, { q: q.question, chosen, correct: q.answerIndex, ok }]);
   172→   172→    if (idx + 1 < questions.length) {
   173→   173→      setIdx(idx + 1);
   174→   174→      setChosen(null);
   175→   175→    } else {
   176→   176→      setFinished(true);
   177→   177→    }
   178→   178→  };
   179→   179→
   180→   180→  const score = answers.filter(a => a.ok).length;
   181→   181→  const pct = answers.length ? Math.round((score / answers.length) * 100) : 0;
   182→   182→
   183→   183→  return (
   184→   184→    <div>
   185→   185→      <div className="rounded-2xl border border-border/70 bg-card p-5">
   186→   186→        <div className="flex flex-col gap-3 sm:flex-row sm:items-end">
   187→   187→          <div className="flex-1 space-y-1.5">
   188→   188→            <label className="text-xs font-medium text-muted-foreground">Scope</label>
   189→   189→            <SkillScopeSelect value={scope} onChange={setScope} />
   190→   190→          </div>
   191→   191→          <div className="space-y-1.5">
   192→   192→            <label className="text-xs font-medium text-muted-foreground">Difficulty</label>
   193→   193→            <Select value={difficulty} onValueChange={setDifficulty}>
   194→   194→              <SelectTrigger className="w-full sm:w-40"><SelectValue /></SelectTrigger>
   195→   195→              <SelectContent>
   196→   196→                <SelectItem value="easy">Easy — recall</SelectItem>
   197→   197→                <SelectItem value="medium">Medium — apply</SelectItem>
   198→   198→                <SelectItem value="hard">Hard — judge</SelectItem>
   199→   199→                <SelectItem value="mixed">Mixed</SelectItem>
   200→   200→              </SelectContent>
   201→   201→            </Select>
   202→   202→          </div>
   203→   203→          <Button onClick={() => void generate()} disabled={loading} className="gap-2">
   204→   204→            {loading ? <Loader2 className="h-4 w-4 animate-spin" /> : <Sparkles className="h-4 w-4" />}
   205→   205→            {loading ? 'Writing exam…' : 'Generate quiz'}
   206→   206→          </Button>
   207→   207→        </div>
   208→   208→        <p className="mt-3 text-xs text-muted-foreground">
   209→   209→          Five CBAP-style multiple-choice questions written fresh by your coach every time.
   210→   210→        </p>
   211→   211→      </div>
   212→   212→
   213→   213→      {loading && (
   214→   214→        <div className="mt-6 flex items-center gap-3 rounded-xl border border-border/60 bg-card p-4 text-sm text-muted-foreground">
   215→   215→          <Loader2 className="h-4 w-4 animate-spin text-primary" /> Your coach is writing fresh exam questions…
   216→   216→        </div>
   217→   217→      )}
   218→   218→
   219→   219→      {questions && questions.length === 0 && !loading && (
   220→   220→        <div className="mt-6 rounded-xl border border-destructive/40 bg-destructive/10 p-4 text-sm">
   221→   221→          Quiz generation failed. Please try again.
   222→   222→        </div>
   223→   223→      )}
   224→   224→
   225→   225→      {questions && questions.length > 0 && !finished && (
   226→   226→        <div className="mt-6 rounded-2xl border border-border/70 bg-card p-6">
   227→   227→          <div className="mb-4 flex items-center justify-between text-xs text-muted-foreground">
   228→   228→            <span>Question {idx + 1} of {questions.length}</span>
   229→   229→            <span>Score {score}/{answers.length}</span>
   230→   230→          </div>
   231→   231→          <div className="mb-2 h-1 w-full overflow-hidden rounded-full bg-muted">
   232→   232→            <div className="h-full rounded-full bg-primary transition-all" style={{ width: `${((idx + (chosen !== null ? 1 : 0)) / questions.length) * 100}%` }} />
   233→   233→          </div>
   234→   234→          <h3 className="mb-4 mt-4 text-base font-medium leading-snug">{questions[idx].question}</h3>
   235→   235→          <div className="space-y-2">
   236→   236→            {questions[idx].options.map((opt, i) => (
   237→   237→              <button
   238→   238→                key={i}
   239→   239→                onClick={() => setChosen(i)}
   240→   240→                className={cn(
   241→   241→                  'flex w-full items-start gap-3 rounded-xl border p-3 text-left text-sm transition',
   242→   242→                  chosen === i
   243→   243→                    ? 'border-primary bg-primary/10'
   244→   244→                    : 'border-border/70 hover:border-primary/40 hover:bg-accent/40'
   245→   245→                )}
   246→   246→              >
   247→   247→                <span className={cn(
   248→   248→                  'flex h-6 w-6 shrink-0 items-center justify-center rounded-full border text-xs font-semibold',
   249→   249→                  chosen === i ? 'border-primary bg-primary text-primary-foreground' : 'border-border text-muted-foreground'
   250→   250→                )}>
   251→   251→                  {String.fromCharCode(65 + i)}
   252→   252→                </span>
   253→   253→                {opt}
   254→   254→              </button>
   255→   255→            ))}
   256→   256→          </div>
   257→   257→          <Button className="mt-5 w-full" onClick={submit} disabled={chosen === null}>
   258→   258→            {idx + 1 === questions.length ? 'Finish quiz' : 'Next question'}
   259→   259→          </Button>
   260→   260→        </div>
   261→   261→      )}
   262→   262→
   263→   263→      {finished && questions && (
   264→   264→        <div className="mt-6 rounded-2xl border border-border/70 bg-card p-6">
   265→   265→          <div className="mb-4 flex items-center gap-4">
   266→   266→            <div className={cn(
   267→   267→              'flex h-16 w-16 items-center justify-center rounded-2xl text-lg font-bold',
   268→   268→              pct >= 80 ? 'bg-primary/15 text-primary' : pct >= 50 ? 'bg-amber-500/15 text-amber-600 dark:text-amber-400' : 'bg-destructive/15 text-destructive'
   269→   269→            )}>
   270→   270→              {pct}%
   271→   271→            </div>
   272→   272→            <div>
   273→   273→              <h3 className="text-lg font-semibold">
   274→   274→                {pct >= 80 ? 'Outstanding — coach level.' : pct >= 50 ? 'Solid — keep drilling.' : 'Worth revisiting this skill.'}
   275→   275→              </h3>
   276→   276→              <p className="text-sm text-muted-foreground">You scored {score} out of {questions.length}.</p>
   277→   277→            </div>
   278→   278→          </div>
   279→   279→          <div className="mb-5 space-y-3">
   280→   280→            {answers.map((a, i) => (
   281→   281→              <div key={i} className={cn('rounded-xl border p-3 text-sm', a.ok ? 'border-primary/30 bg-primary/5' : 'border-destructive/30 bg-destructive/5')}>
   282→   282→                <div className="mb-1 flex items-start gap-2 font-medium">
   283→   283→                  {a.ok ? <CheckCircle2 className="mt-0.5 h-4 w-4 shrink-0 text-primary" /> : <XCircle className="mt-0.5 h-4 w-4 shrink-0 text-destructive" />}
   284→   284→                  {a.q}
   285→   285→                </div>
   286→   286→                {!a.ok && (
   287→   287→                  <p className="pl-6 text-xs text-muted-foreground">
   288→   288→                    Correct: <span className="font-medium text-foreground">{String.fromCharCode(65 + a.correct)}. {questions[i]?.options[a.correct]}</span>
   289→   289→                    {questions[i]?.explanation ? ` — ${questions[i].explanation}` : ''}
   290→   290→                  </p>
   291→   291→                )}
   292→   292→              </div>
   293→   293→            ))}
   294→   294→          </div>
   295→   295→          <Button variant="outline" onClick={() => void generate()} className="gap-2">
   296→   296→            <RotateCcw className="h-4 w-4" /> Try a fresh quiz
   297→   297→          </Button>
   298→   298→        </div>
   299→   299→      )}
   300→   300→    </div>
   301→   301→  );
   302→   302→}

   305→   305→
   306→   306→function FlashcardsTab() {
   307→   307→  const [scope, setScope] = useState('mixed');
   308→   308→  const [cards, setCards] = useState<Flashcard[] | null>(null);
   309→   309→  const [loading, setLoading] = useState(false);
   310→   310→  const [idx, setIdx] = useState(0);
   311→   311→  const [flipped, setFlipped] = useState(false);
   312→   312→  const [known, setKnown] = useState<Set<number>>(new Set());
   313→   313→
   314→   314→  const generate = async () => {
   315→   315→    setLoading(true);
   316→   316→    setCards(null);
   317→   317→    setIdx(0);
   318→   318→    setFlipped(false);
   319→   319→    setKnown(new Set());
   320→   320→    try {
   321→   321→      const isCategory = ['mixed', ...Object.keys(CATEGORY_META)].includes(scope);
   322→   322→      const res = await apiFetch('/api/flashcards', {
   323→   323→        method: 'POST',
   324→   324→        body: JSON.stringify({
   325→   325→          skillSlug: isCategory ? undefined : scope,
   326→   326→          category: isCategory ? scope : undefined,
   327→   327→          count: 8,
   328→   328→        }),
   329→   329→      });
   330→   330→      const data = await readJson<{ error?: string; cards?: Flashcard[] }>(res);
   331→   331→      if (!res.ok || data.error) throw new Error(data.error);
   332→   332→      setCards(data.cards || []);
   333→   333→    } catch {
   334→   334→      setCards([]);
   335→   335→    } finally {
   336→   336→      setLoading(false);
   337→   337→    }
   338→   338→  };
   339→   339→
   340→   340→  const mark = (knew: boolean) => {
   341→   341→    if (!cards) return;
   342→   342→    const next = new Set(known);
   343→   343→    if (knew) next.add(idx);
   344→   344→    else next.delete(idx);
   345→   345→    setKnown(next);
   346→   346→    if (idx + 1 < cards.length) {
   347→   347→      setIdx(idx + 1);
   348→   348→      setFlipped(false);
   349→   349→    }
   350→   350→  };
   351→   351→
   352→   352→  return (
   353→   353→    <div>
   354→   354→      <div className="rounded-2xl border border-border/70 bg-card p-5">
   355→   355→        <div className="flex flex-col gap-3 sm:flex-row sm:items-end">
   356→   356→          <div className="flex-1 space-y-1.5">
   357→   357→            <label className="text-xs font-medium text-muted-foreground">Deck scope</label>
   358→   358→            <SkillScopeSelect value={scope} onChange={setScope} />
   359→   359→          </div>
   360→   360→          <Button onClick={() => void generate()} disabled={loading} className="gap-2">
   361→   361→            {loading ? <Loader2 className="h-4 w-4 animate-spin" /> : <Layers className="h-4 w-4" />}
   362→   362→            {loading ? 'Preparing deck…' : 'Generate flashcards'}
   363→   363→          </Button>
   364→   364→        </div>
   365→   365→        <p className="mt-3 text-xs text-muted-foreground">Spaced-repetition style cards, freshly generated per skill.</p>
   366→   366→      </div>
   367→   367→
   368→   368→      {loading && (
   369→   369→        <div className="mt-6 flex items-center gap-3 rounded-xl border border-border/60 bg-card p-4 text-sm text-muted-foreground">
   370→   370→          <Loader2 className="h-4 w-4 animate-spin text-primary" /> Preparing your deck…
   371→   371→        </div>
   372→   372→      )}
   373→   373→
   374→   374→      {cards && cards.length > 0 && (
   375→   375→        <div className="mt-6">
   376→   376→          <div className="mb-3 flex items-center justify-between text-xs text-muted-foreground">
   377→   377→            <span>Card {idx + 1} of {cards.length}</span>
   378→   378→            <span className="flex items-center gap-1 text-primary"><CheckCircle2 className="h-3.5 w-3.5" /> {known.size} known</span>
   379→   379→          </div>
   380→   380→          <div className="flip-scene h-64 w-full" onClick={() => setFlipped(f => !f)} role="button" aria-label="Flip card">
   381→   381→            <div className={cn('flip-inner', flipped && 'flipped')}>
   382→   382→              <div className="flip-face flex cursor-pointer flex-col items-center justify-center rounded-2xl border border-border/70 bg-card p-8 text-center shadow-sm">
   383→   383→                <Badge variant="outline" className="mb-4 text-[10px]">PROMPT</Badge>
   384→   384→                <p className="text-lg font-medium leading-snug">{cards[idx].front}</p>
   385→   385→                <span className="mt-6 flex items-center gap-1.5 text-xs text-muted-foreground">
   386→   386→                  <FlipHorizontal2 className="h-3.5 w-3.5" /> Click to reveal
   387→   387→                </span>
   388→   388→              </div>
   389→   389→              <div className="flip-face flip-face-back flex cursor-pointer flex-col items-center justify-center rounded-2xl border border-primary/40 bg-primary/8 p-8 text-center shadow-sm">
   390→   390→                <Badge variant="outline" className="mb-4 text-[10px]">ANSWER</Badge>
   391→   391→                <p className="text-sm leading-relaxed">{cards[idx].back}</p>
   392→   392→              </div>
   393→   393→            </div>
   394→   394→          </div>
   395→   395→          <div className="mt-4 flex items-center justify-center gap-3">
   396→   396→            <Button variant="outline" className="gap-2" onClick={() => mark(false)}>
   397→   397→              <RotateCcw className="h-4 w-4" /> Review again

   400→   400→              <CheckCircle2 className="h-4 w-4" /> I knew this
   401→   401→            </Button>
   402→   402→          </div>
   403→   403→          <div className="mt-4 flex justify-center gap-2">
   404→   404→            <Button variant="ghost" size="icon" disabled={idx === 0} onClick={() => { setIdx(idx - 1); setFlipped(false); }} aria-label="Previous card">

   406→   406→            </Button>
   407→   407→            <Button variant="ghost" size="icon" disabled={idx === cards.length - 1} onClick={() => { setIdx(idx + 1); setFlipped(false); }} aria-label="Next card">
   408→   408→              <ChevronRight className="h-4 w-4" />
   409→   409→            </Button>
   410→   410→          </div>
   411→   411→        </div>
   412→   412→      )}
   413→   413→
   414→   414→      {cards && cards.length === 0 && !loading && (
   415→   415→        <div className="mt-6 rounded-xl border border-destructive/40 bg-destructive/10 p-4 text-sm">Card generation failed. Please try again.</div>
